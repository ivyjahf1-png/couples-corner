
=== messages (required: ["id","sender_id","content","created_at","type","status","updated_at"]) ===
  id                     string     uuid                     not-null default="gen_random_uuid()"
  sender_id              string     uuid                     not-null default=null
  content                string     text                     not-null default=null
  created_at             string     timestamp with time zone not-null default="timezone('utc'::text, now())"
  conversation_id        string     uuid                     not-null default=null
  type                   string     text                     not-null default="text"
  body                   string     text                     not-null default=null
  status                 string     text                     not-null default="sent"
  read_at                string     timestamp with time zone not-null default=null
  updated_at             string     timestamp with time zone not-null default="now()"

=== conversations (required: ["id","type","participant_user_ids","created_at","updated_at"]) ===
  id                     string     uuid                     not-null default="gen_random_uuid()"
  type                   string     text                     not-null default="direct"
  participant_user_ids   array      uuid[]                   not-null default=null
  created_by_id          string     uuid                     not-null default=null
  last_message_at        string     timestamp with time zone not-null default=null
  last_message_id        string     uuid                     not-null default=null
  created_at             string     timestamp with time zone not-null default="now()"
  updated_at             string     timestamp with time zone not-null default="now()"
